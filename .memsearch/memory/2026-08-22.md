// kimi — the inverse-kimi TELEPORTER. miki is provider-free: the cloud agent
// ITSELF is the provider, and `kimi` is how that agent reaches the sandbox
// without ever cloning the repo to its own machine.
//
// THE PROBLEM THIS KILLS: a cloud agent that downloads the whole repo, edits
// locally, then uploads the diff. That round-trip is slow, lossy, and breaks on
// symlinks / build artifacts. Instead, `kimi` teleports the agent's commands
// straight into the workspace: `kimi s ls` lists the box, `kimi s "cargo check"`
// builds on the box, `kimi s git ...` operates on the box repo. The agent never
// holds the code — it holds a shell into the box.
//
// Credentials are set ONCE, as env (the delegator supplies them — nothing is
// baked in):
//   SSH overlay (preferred — direct, no proxy, no per-call tokens):
//     MIKI_SSH_HOST    box hostname / IP        (e.g. 34.72.35.255)
//     MIKI_SSH_PORT    ssh port                 (default 22)
//     MIKI_SSH_USER    login user               (e.g. agent)
//     MIKI_SSH_KEY     path to ed25519 key  OR
//     MIKI_SSH_KEY_B64 base64 of that key (no file needed)

//     MIKI_SID       Daytona toolbox session id

#!/bin/bash

set -e

# Boot timeline observability: lets us attribute FC 412 (port 81 health check
# failed in 120s) to image-pull vs start.sh-step latency. /proc/uptime gives
# seconds since container start, so subtracting from current time isolates the
# FC platform overhead from anything start.sh itself does.
BOOT_TIMELINE=/tmp/boot-timeline.log
: > "$BOOT_TIMELINE"
# All helpers below guard /proc/uptime with `|| echo ?` so a missing file
# (non-Linux dev shells, broken procfs) never trips set -e and kills bootstrap.
boot_uptime() { awk '{print $1}' /proc/uptime 2>/dev/null || echo "?"; }
boot_ts()     { date -u +%FT%TZ; }
echo "[BOOT] start.sh entered at $(boot_ts) uptime=$(boot_uptime)s" | tee -a "$BOOT_TIMELINE"

# Dump the full timeline on any exit. FC's 412 error body only captures stdout
# tail; without this, all early-step durations get truncated and we lose the
# evidence we need to pick a real fix.
dump_boot_timeline() {
    local rc=$?
    echo "[BOOT] exit rc=$rc at $(boot_ts) uptime=$(boot_uptime)s" >> "$BOOT_TIMELINE"
    echo ""
    echo "==== BOOT TIMELINE (full) ===="
    cat "$BOOT_TIMELINE" 2>/dev/null || true
    echo "==== END BOOT TIMELINE ===="
}
trap dump_boot_timeline EXIT

"""SSH helper for talking to the work box via paramiko.

Usage:
  python ssh_run.py "<command>"
  python ssh_run.py --read <remote_path>
"""
import sys
import os
import paramiko

KEY_PATH = os.path.expanduser("~/.ssh/id_ed25519")
HOST = "34.72.35.255"
PORT = 2222
USER = "agent"


def get_client():
    client = paramiko.SSHClient()
    client.set_missing_host_key_policy(paramiko.AutoAddPolicy())
    pkey = paramiko.Ed25519Key.from_private_key_file(KEY_PATH)
    client.connect(
        hostname=HOST,
        port=PORT,
        username=USER,
        pkey=pkey,
        look_for_keys=False,
        allow_agent=False,
        timeout=30,
    )
    return client


def run(cmd: str, timeout: int = 600):
    client = get_client()
    try:
        stdin, stdout, stderr = client.exec_command(cmd, timeout=timeout, get_pty=False)
        out = stdout.read().decode("utf-8", errors="replace")
        err = stderr.read().decode("utf-8", errors="replace")
        rc = stdout.channel.recv_exit_status()
        sys.stdout.write(out)
        if err:
            sys.stderr.write(err)
        sys.exit(rc)
    finally:
        client.close()


def read_file(remote_path: str):
    client = get_client()
    try:
        sftp = client.open_sftp()
        with sftp.open(remote_path, "r") as f:
            data = f.read()
        sys.stdout.buffer.write(data)
    finally:
        client.close()


if __name__ == "__main__":
    args = sys.argv[1:]
    if not args:
        print("usage: ssh_run.py <cmd> | --read <path>", file=sys.stderr)
        sys.exit(2)
    if args[0] == "--read":
        read_file(args[1])
    else:
        cmd = " ".join(args)
        run(cmd)
